
-d contains cos
SELECT classes.classid, crosslistings.dept, crosslistings.coursenum, courses.area, courses.title
from classes
inner join courses 
on classes.courseid = courses.courseid
inner join crosslistings 
on crosslistings.courseid = classes.courseid 
where crosslistings.dept like "%cos%"
order by crosslistings.dept, crosslistings.coursenum, classes.classid
